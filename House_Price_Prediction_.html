Select (X) and (y)

feature_cols = [

Evaluation

In [49]:
# Predictions
y_train_pred = model.predict(X_train_scaled)
y_test_pred = model.predict(X_test_scaled)

# Return to original price from logs (to be valued at $)
y_test_real = np.expm1(y_test)
y_test_pred_real = np.expm1(y_test_pred)

# Metrics
print("TRAIN")
print("R2 :", round(r2_score(y_train, y_train_pred), 4))
print("TEST")
print("R2  :", round(r2_score(y_test, y_test_pred), 4))
rmse = np.sqrt(mean_squared_error(y_test_real, y_test_pred_real))
mae = mean_absolute_error(y_test_real, y_test_pred_real)
print(f"RMSE: ${rmse:,.0f}")
print(f"MAE : ${mae:,.0f}")
TRAIN R2 : 0.6475 TEST R2 : 0.6521 RMSE: $284,284 MAE : $135,898
 
R2 is the percentage of price elasticity the model explains (close to 1 = good).
RMSE / MAE is the average error (in dollars, small = good).
If Train and Test R2 are close, the model is not overfitted.
